```python
from abc import ABC, abstractmethod

class DataCollector(ABC):
    """추상 데이터 수집 클래스"""
    
    @abstractmethod
    def collect(self) -> pd.DataFrame:
        """데이터 수집"""
        pass
    
    @abstractmethod
    def validate(self, df: pd.DataFrame) -> bool:
        """데이터 검증"""
        pass

# 5개 프로그램별 Collector 구현
class 입고정보Collector(DataCollector): ...
class 출고정보Collector(DataCollector): ...
class 재고정보Collector(DataCollector): ...
class 삭제정보Collector(DataCollector): ...
class 비정형오더Collector(DataCollector): ...

# 통합 관리
class DataIntegrator:
    """모든 데이터 소스 통합 관리"""
    
    def collect_all(self) -> Dict[str, pd.DataFrame]:
        """전체 데이터 수집"""
        pass
```

### 동적 설정 로드

```python
# src/utils/config.py

import yaml
from typing import Any, Dict

class Config:
    """설정 관리 싱글톤"""
    
    _instance = None
    _config: Dict[str, Any] = {}
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._load_config()
        return cls._instance
    
    def _load_config(self):
        """YAML 설정 파일 로드"""
        with open('config/config.yaml', 'r', encoding='utf-8') as f:
            self._config = yaml.safe_load(f)
    
    def get(self, key: str, default: Any = None) -> Any:
        """설정값 조회"""
        keys = key.split('.')
        value = self._config
        for k in keys:
            value = value.get(k, default)
            if value is default:
                break
        return value

# 사용 예시
config = Config()
refresh_interval = config.get('refresh_interval', 30)
```

```python
# src/utils/error_handler.py

import logging
from functools import wraps
from typing import Callable

logger = logging.getLogger(__name__)

def handle_data_error(func: Callable):
    """데이터 수집/처리 에러 핸들링"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        try:
            return func(*args, **kwargs)
        except FileNotFoundError as e:
            logger.error(f"데이터 파일 없음: {e}")
            return pd.DataFrame()  # 빈 DataFrame 반환
        except pd.errors.ParserError as e:
            logger.error(f"데이터 파싱 오류: {e}")
            return pd.DataFrame()
        except Exception as e:
            logger.exception(f"예상치 못한 오류: {e}")
            return pd.DataFrame()
    return wrapper

def handle_ui_error(func: Callable):
    """UI 렌더링 에러 핸들링"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        try:
            return func(*args, **kwargs)
        except Exception as e:
            logger.exception(f"UI 렌더링 오류: {e}")
            st.error(f"⚠️ 화면 표시 중 오류가 발생했습니다: {str(e)}")
    return wrapper
```

```python
# src/utils/metrics_collector.py

class PerformanceMetrics:
    """성능 메트릭 수집"""
    
    def __init__(self):
        self.metrics = {
            'data_load_time': [],
            'calculation_time': [],
            'render_time': [],
            'total_time': []
        }
    
    def record(self, metric_name: str, value: float):
        """메트릭 기록"""
        self.metrics[metric_name].append(value)
    
    def get_average(self, metric_name: str) -> float:
        """평균값 조회"""
        values = self.metrics.get(metric_name, [])
        return sum(values) / len(values) if values else 0
```
