## 개요

### 목적
5개 핵심 모듈에서 수집한 실제 데이터를 v9 HTML 대시보드에 적용하여 운영 환경에 배포

### 핵심 원칙
1. **절대 추측하지 않기**: 모든 데이터는 실제 CSV에서 확인
2. **용어 통일**: 케로님의 용어 사용 (출고 타입 설명.txt 기준)
3. **⭐ 카드6 배송처 분류**: 출고유형 + 배송군 조합으로만 분류 (절대 추측 금지!)
4. **단계별 검증**: 각 단계마다 결과 확인 후 다음 단계 진행
5. **문서 우선**: 참고 문서를 먼저 읽고 작업 시작

### 5개 핵심 모듈
```
1. 입고 현황 (InboundCollector)
   → 카드2 데이터

2. 출고 현황 (OutboundCollector)
   → 카드5, 카드6 데이터
```

```python
import pandas as pd
from datetime import datetime

today = datetime.now().strftime("%Y%m%d")
inbound = pd.read_csv(
    f'C:/OSIS_AUTO/Inbound Status/integrated_inbound_{today}.csv',
    encoding='utf-8-sig'
)

# 1. 총 건수
total_count = len(inbound)

# 2. 진척률 (평균)
progress_rate = inbound['진척률'].mean()

# 3. 입고유의상품 (입고 소비기한 < 재고 소비기한)
inventory = pd.read_csv(
    f'C:/OSIS_AUTO/inventory_status/inventory_status_{today}.csv',
    encoding='utf-8-sig'
)

# 입고 상품별 최소 소비기한
inbound_expiry = inbound.groupby('상품')['소비기한'].min().reset_index()
inbound_expiry.columns = ['상품', '입고_소비기한']

# 재고 상품별 최소 소비기한
inventory_expiry = inventory.groupby('상품')['소비기한'].min().reset_index()
inventory_expiry.columns = ['상품', '재고_소비기한']

# 교집합
merged = pd.merge(inbound_expiry, inventory_expiry, on='상품', how='inner')

# 입고 소비기한이 재고 소비기한보다 빠른 경우
risky = merged[merged['입고_소비기한'] < merged['재고_소비기한']]
inbound_risky_count = len(risky)

print(f"totalCount: {total_count}")
print(f"progressRate: {progress_rate:.1f}")
print(f"inboundRiskyCount: {inbound_risky_count}")
```

```python
import pandas as pd

inventory = pd.read_csv(
    f'C:/OSIS_AUTO/inventory_status/inventory_status_{today}.csv',
    encoding='utf-8-sig'
)

# 1. 유효유통비 20% 이하 필터링
risky = inventory[inventory['유효유통비(%)'] <= 20].copy()

# 2. L07 로케이션 제외 ⭐ 중요!
risky_filtered = risky[~risky['로케이션'].str.startswith('L07')].copy()

# 3. 긴급/주의 구분
urgent_count = len(risky_filtered[risky_filtered['유효유통비(%)'] <= 10])
warning_count = len(risky_filtered[risky_filtered['유효유통비(%)'] > 10])

total_count = len(risky_filtered)

# 4. 상세 데이터 (유효비 오름차순)
risky_filtered = risky_filtered.sort_values('유효유통비(%)')

# 5. JSON 형식 변환
items_list = []
for _, row in risky_filtered.iterrows():
    item = {
        "location": row['로케이션'],
        "productCode": str(row['상품']),
        "productName": row['상품명'],
        "expiryDate": str(row['소비기한']),
        "stockQty": int(row['재고수량']),
        "validityRatio": float(row['유효유통비(%)'])
    }
    items_list.append(item)

print(f"totalCount: {total_count}")
print(f"urgentCount: {urgent_count}")
print(f"warningCount: {warning_count}")
print(f"riskyItems: {len(items_list)}개")
```

```python
    print(f"❌ 전일 대비 파일 없음")

print(f"amount: {amount}")
print(f"compare: {compare:.1f}")
print(f"totalLabel: {total_label}")
```

```python
def classify_destination_card6(row):
    """
    카드6 배송처 분류 로직
    
    ⚠️ 절대 추측하지 마세요!
    ⚠️ 출고유형 + 배송군 조합으로만 분류!
    ⚠️ 배송처명은 마지막 fallback!
    """
    출고유형 = row['출고유형']
    배송군_str = str(row['배송군'])
    배송처명 = row['배송처명']
    
    배송군_4자리 = len(배송군_str) == 4
    
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    if 출고유형 == 5:
        if 배송처명 == '식재':
            return '한익스'
        elif 배송군_4자리 and 배송군_str[0] == '3':
            return '키즈'
        elif 배송군_4자리 and 배송군_str[0] == '4':
            return '용인3'
        elif 배송군_4자리 and 배송군_str[0] == '2':
            return '용인2'
    
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    # 규칙 2: 08 타입
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    elif 출고유형 == 8:
        if 배송군_4자리 and 배송군_str[0] == '4':
            return '용인3'
        elif 배송군_4자리 and 배송군_str[0] == '2':
            return '용인2'
    
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    # 규칙 3: 기타 지방 출고 (04, 16, 17, 52, 53)
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    elif 출고유형 in [4, 16, 17, 52, 53]:
        if '용인' in 배송처명:
            if 배송군_4자리 and 배송군_str[0] == '4':
                return '용인3'
            else:
                return '용인2'
        elif 배송처명 == '양산':
            return '양산'
        elif 배송처명 == '양산2':
            return '양산2'
        else:
            # 제주, 제천, 호남, 구미, 계룡, 음성, 안산 등
            return 배송처명
    
    return 배송처명
```

- **프로그램은 실행되지만 파일은 업데이트 안 됨!**

**확인 방법**:
```powershell
# CSV 파일 열고 있는 프로세스 찾기
Get-Process | Where-Object {
    $_.MainWindowTitle -like "*outbound*.csv*" -or
    $_.MainWindowTitle -like "*inbound*.csv*" -or
    $_.MainWindowTitle -like "*inventory*.csv*"
} | Select-Object ProcessName, Id, MainWindowTitle

# 출력 예시:
# ProcessName    Id MainWindowTitle                  
# -----------    -- ---------------                  
# EXCEL       11332 outbound_all_20251116.csv - Excel
```

**해결**:
1. **방법 1**: 해당 파일을 닫기 (안전)
2. **방법 2**: Excel 프로세스 종료 (주의! 다른 파일도 닫힘)
3. 백엔드 프로그램 재실행

```python
import subprocess

def check_file_lock():
```

```python
def verify_backend_execution(file_path, max_age_minutes=10):
    """백엔드 실행 후 파일이 실제로 업데이트되었는지 검증"""
    if not os.path.exists(file_path):
        raise FileNotFoundError(f"파일 없음: {file_path}")
    
    mtime = datetime.fromtimestamp(os.path.getmtime(file_path))
    age_minutes = (datetime.now() - mtime).total_seconds() / 60
    
    if age_minutes > max_age_minutes:
        raise Exception(
            f"파일이 오래되었습니다!\n"
            f"수정 시각: {mtime}\n"
            f"경과 시간: {age_minutes:.0f}분\n"
            f"→ 백엔드 재실행 또는 파일 잠금 확인 필요"
        )
    
    return True

# 사용 예시
verify_backend_execution("C:/OSIS_AUTO/Outbound Status/outbound_all_20251116.csv")
```
