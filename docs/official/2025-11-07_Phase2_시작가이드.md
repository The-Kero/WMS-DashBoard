## 🎯 Phase 2 개요

### 목표

```
✅ Flask API 서버 (REST API)

✅ 30초 자동 갱신 (AJAX)

```

| 항목 | 목표 | 측정 방법 |
|------|------|----------|
| 페이지 로딩 | < 2초 | 브라우저 개발자 도구 |

## 🔄 Collector 재사용 전략

### Collector 임포트 방법

```python
# flask_app/services/collector_service.py

import sys
from pathlib import Path
from datetime import datetime

# Collector 경로 추가
collector_path = Path(__file__).parent.parent.parent / "dashboard" / "src" / "data" / "collectors"
sys.path.insert(0, str(collector_path))

# Collector 임포트
from inbound import InboundCollector
from outbound import OutboundCollector
from inventory import InventoryCollector
from delete import DeleteCollector
from irregular import IrregularCollector


class CollectorService:
    """Collector 관리 서비스"""
```

## ⏱️ 30초 자동 갱신 구현

### dashboard.js

```javascript
// static/js/dashboard.js

class Dashboard {
    constructor() {
        this.apiUrl = '/api/dashboard';
        this.refreshInterval = 30000; // 30초
        this.timer = null;
        this.isLoading = false;
    }

    /**
     * 초기화
     */
    init() {
        console.log('Dashboard 초기화...');
        this.loadData(); // 첫 로딩
        this.startAutoRefresh(); // 자동 갱신 시작
    }

    /**
     * 데이터 로드
     */
    async loadData() {
        if (this.isLoading) return;
        
        this.isLoading = true;
        console.log('데이터 로딩 중...');
        
        try {
            const response = await fetch(this.apiUrl);
            const data = await response.json();
            
            if (data.success) {
                this.updateCards(data.cards);
                this.updateTimestamp();
                console.log('데이터 로딩 완료');
            } else {
                console.error('데이터 로딩 실패:', data.error);
                this.showError('데이터를 불러올 수 없습니다.');
            }
        } catch (error) {
            console.error('API 호출 오류:', error);
            this.showError('서버와 통신할 수 없습니다.');
        } finally {
            this.isLoading = false;
        }
    }

    /**
     * 카드 업데이트
     */
    updateCards(cards) {
        cards.forEach(card => {
            // 값 업데이트
            const valueElement = document.getElementById(card.id);
            if (valueElement) {
                valueElement.textContent = this.formatNumber(card.value);
                this.animateValue(valueElement);
            }
            
            // 증감률 업데이트
            const changeElement = document.getElementById(`${card.id}-change`);
            if (changeElement) {
                changeElement.textContent = card.change;
                changeElement.className = `card-change trend-${card.trend}`;
            }
        });
    }

    /**
     * 숫자 포맷팅
     */
    formatNumber(value) {
        if (typeof value === 'number') {
            return value.toLocaleString('ko-KR');
        }
        return value;
    }

    /**
     * 값 변경 애니메이션
     */
    animateValue(element) {
        element.classList.add('value-updated');
        setTimeout(() => {
            element.classList.remove('value-updated');
        }, 500);
    }

    /**
     * 타임스탬프 업데이트
     */
    updateTimestamp() {
        const now = new Date();
        const timeString = now.toLocaleTimeString('ko-KR');
        document.getElementById('last-update-time').textContent = timeString;
    }

    /**
     * 자동 갱신 시작
     */
    startAutoRefresh() {
        this.timer = setInterval(() => {
            this.loadData();
        }, this.refreshInterval);
        console.log(`자동 갱신 시작 (${this.refreshInterval / 1000}초 주기)`);
    }

    /**
     * 자동 갱신 중지
     */
    stopAutoRefresh() {
        if (this.timer) {
            clearInterval(this.timer);
            this.timer = null;
            console.log('자동 갱신 중지');
        }
    }

    /**
     * 에러 표시
     */
    showError(message) {
        // TODO: 에러 모달 또는 알림 표시
        alert(message);
    }
}

// Dashboard 인스턴스 생성 및 초기화
const dashboard = new Dashboard();
dashboard.init();
```
